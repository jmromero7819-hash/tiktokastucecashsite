VÉRIFICATION DU RENDU CREATOMATE // ======================================== document.getElementById("checkRenderStatus").onclick = async () => { const output = document.getElementById("renderCheckStatus"); const renderStatus = document.getElementById("renderStatus"); const renderUrl = document.getElementById("renderUrl"); const debug = document.getElementById("creatomateDebug"); if (!lastRenderId) { output.innerText = "Aucun Render ID disponible."; return; } output.innerText = "Vérification du rendu..."; try { const response = await fetch("/api/render-status", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ render_id: lastRenderId }) }); const data = await response.json(); // IMPORTANT : // remplace l'ancien JSON par la réponse actuelle debug.innerText = JSON.stringify(data, null, 2); if (!response.ok || !data.success) { output.innerText = data.error || "Erreur pendant la vérification Creatomate."; return; } let render = data.render; if (Array.isArray(render)) { render = render[0]; } if (render && render.data) { render = render.data; } if (!render) { output.innerText = "Aucune information de rendu reçue."; return; } const currentStatus = render.status || "Statut inconnu"; const url = render.url || render.output_url || ""; const errorMessage = render.error_message || render.error || ""; renderStatus.innerText = currentStatus; renderUrl.innerText = url || "Pas encore disponible"; if (currentStatus === "succeeded" && url) { document.getElementById("videoUrl").value = url; output.innerText = "Vidéo terminée. L'URL a été ajoutée automatiquement."; } else if (currentStatus === "failed") { output.innerText = errorMessage ? "Échec Creatomate : " + errorMessage : "Le rendu Creatomate a échoué. Consulte le JSON ci-dessous."; } else { output.innerText = "Le rendu est encore en cours : " + currentStatus; } } catch (error) { output.innerText = "Erreur pendant la vérification du rendu."; debug.innerText = error.message; console.error(error); } };
